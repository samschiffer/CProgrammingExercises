a = (a / b) * b + a % b
a % b = a - (a / b) * b
-8 % 5 = -8 - (-8 / 5) * 5

(b)
       = -8 - (-1) * 5
       = -8 - (-5)
       = -3
OR
       = -8 - (-2) * 5
       = -8 - (-10)
       = -8 + 10
       = 2
(c)
8 % -5 = 8 - (8 / -5) * 5
       = 8 - (-1) * -5
       = 8 - (5)
       = 3
OR
      = 8 - (-2) * (-5)

      = -2
